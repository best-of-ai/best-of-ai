{{/* Picks the recently added tools and orders them by rank (highest first). "Recent" = dated within 45 days of the newest tool; newer date breaks rank ties. Usage: {{ $tools = partial "newest-first.html" $tools }} */}} {{- $byDate := sort . "Date" "desc" -}} {{- $recent := slice -}} {{- if $byDate -}} {{- $cutoff := (index $byDate 0).Date.AddDate 0 0 -45 -}} {{- $recent = where $byDate "Date" "ge" $cutoff -}} {{- end -}} {{- return (sort $recent "Params.rank" "desc") -}}
